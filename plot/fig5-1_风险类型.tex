\newcommand{\riskcol}[8]{%
    \node[anchor=north] (#1-icon) at (#2, 0) {\tikz[scale=1.15]{
        \fill[#3!14, rounded corners=3pt] (-0.34,-0.34) rectangle (0.34,0.34);
        \begin{scope}[draw=#3!90!black, thick, line join=round, line cap=round] #4 \end{scope}}};
    \node[below=3pt of #1-icon, font=\small\bfseries] (#1-id) {#5};
    \node[below=0pt of #1-id, font=\footnotesize\bfseries] (#1-name) {#6};
    \coordinate (#1-rule) at ($(#1-name.south) + (0, -0.16)$);
    \draw[#3!85!black, line width=0.4pt] ($(#1-rule) + (-1.55, 0)$) -- ($(#1-rule) + (1.55, 0)$);
    \node[lab=#3, below=4pt of #1-rule] (#1-l1) {潜在后果};
    \node[txt, below=0pt of #1-l1] (#1-c) {#7};
    \node[lab=#3, below=3pt of #1-c] (#1-l2) {期望响应};
    \node[txt, below=0pt of #1-l2] (#1-r) {#8};
}
